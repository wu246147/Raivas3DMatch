cmake_minimum_required(VERSION 3.21)
project(SurfaceMatchCpp VERSION 1.0 LANGUAGES C CXX)
if(POLICY CMP0144)
    cmake_policy(SET CMP0144 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SM_BUILD_GUI "构建 Qt 测试界面" ON)
option(SM_BUILD_TESTS "构建算法验证程序" ON)
option(SM_ENABLE_RAIVAS "按原生头文件启用 RaiVas 可选 DLL" ON)
option(SM_ENABLE_DEPTH_FILTER "按原生头文件启用深度滤波 DLL" ON)
option(SM_ENABLE_PCL_VIEWER "构建可旋转的 PCL/VTK 点云视图（Windows）" ${WIN32})
option(SM_BUILD_C_API "构建可供 C# P/Invoke 调用的原生 DLL" ON)
set(HALCON_ROOT "$ENV{HALCONROOT}" CACHE PATH "HALCON 安装目录")
set(RAIVAS_ROOT "" CACHE PATH "ocr_alg_yolo_obb_opencv 源码目录")
set(RAIVAS_RUNTIME_DIR "" CACHE PATH "RaiVas DLL 及其依赖所在目录")
set(DEPTH_FILTER_ROOT "" CACHE PATH "pcl_rescale_ply 原始头文件目录")
set(DEPTH_FILTER_RUNTIME_DIR "" CACHE PATH "RaivasAlgDepthFilter DLL 及依赖所在目录")

# Qt Creator 直接打开 CMakeLists.txt 时可能不使用 preset；优先保留调用方显式路径。
if(WIN32 AND NOT OpenCV_DIR)
    find_path(_local_opencv_config OpenCVConfig.cmake
        HINTS "$ENV{OpenCV_DIR}" "D:/opencv4.8.0/opencv/build/x64/vc16/lib"
            "D:/opencv4.8.0/opencv/build" "D:/opencv4.10.0/opencv/build/x64/vc16/lib")
    if(_local_opencv_config)
        set(OpenCV_DIR "${_local_opencv_config}" CACHE PATH "OpenCVConfig.cmake 所在目录" FORCE)
    endif()
endif()
if(WIN32 AND NOT CMAKE_PREFIX_PATH AND EXISTS "D:/Qt/5.15.2/msvc2019_64/lib/cmake/Qt5")
    list(APPEND CMAKE_PREFIX_PATH "D:/Qt/5.15.2/msvc2019_64")
endif()
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs calib3d)
find_path(HALCON_INCLUDE_DIR HalconCpp.h HINTS "${HALCON_ROOT}/include/halconcpp" REQUIRED)
find_library(HALCON_CPP_LIBRARY halconcpp HINTS "${HALCON_ROOT}/lib/x64-win64" REQUIRED)
find_library(HALCON_LIBRARY halcon HINTS "${HALCON_ROOT}/lib/x64-win64" REQUIRED)
add_library(Raivas3DMatch STATIC src/Tool.cpp src/SurfaceMatch.cpp src/NativeBackend.cpp)
add_library(Raivas3DMatch::Raivas3DMatch ALIAS Raivas3DMatch)
# 保留既有 CMake 调用方使用的别名，实际目标和输出文件使用新的库名。
add_library(SurfaceMatch::surface_match ALIAS Raivas3DMatch)
target_include_directories(Raivas3DMatch PUBLIC include "${HALCON_INCLUDE_DIR}" "${HALCON_ROOT}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(Raivas3DMatch PUBLIC ${OpenCV_LIBS} "${HALCON_CPP_LIBRARY}" "${HALCON_LIBRARY}")
target_compile_definitions(Raivas3DMatch PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
    target_compile_options(Raivas3DMatch PUBLIC /utf-8 /Zc:__cplusplus)
    target_compile_options(Raivas3DMatch PRIVATE /W4)
endif()
if(SM_ENABLE_RAIVAS)
    if(NOT WIN32 OR NOT OpenCV_VERSION VERSION_LESS 4.9 OR OpenCV_VERSION VERSION_LESS 4.8)
        message(FATAL_ERROR "已有 RaiVas SDK 要求 Windows x64 / OpenCV 4.8，请使用兼容版本。")
    endif()
    if(NOT EXISTS "${RAIVAS_ROOT}/raivasalgobbdet.h")
        message(FATAL_ERROR "RAIVAS_ROOT 必须指向原始 C++ 头文件目录。")
    endif()
    target_include_directories(Raivas3DMatch PRIVATE "${RAIVAS_ROOT}" "${RAIVAS_ROOT}/3rdparty/alg/include" "${RAIVAS_ROOT}/3rdparty/onnxruntime/include")
    target_compile_definitions(Raivas3DMatch PRIVATE SM_ENABLE_RAIVAS)
endif()
if(SM_ENABLE_DEPTH_FILTER OR (SM_BUILD_GUI AND SM_ENABLE_PCL_VIEWER))
    if(NOT PCL_DIR AND EXISTS "D:/Program Files/PCL 1.12.1/cmake/PCLConfig.cmake")
        set(PCL_DIR "D:/Program Files/PCL 1.12.1/cmake" CACHE PATH "PCLConfig.cmake 所在目录")
    endif()
    set(_pcl_components common)
    if(SM_BUILD_GUI AND SM_ENABLE_PCL_VIEWER)
        list(APPEND _pcl_components visualization)
    endif()
    find_package(PCL REQUIRED COMPONENTS ${_pcl_components})
endif()
if(SM_ENABLE_DEPTH_FILTER)
    target_include_directories(Raivas3DMatch PRIVATE "${DEPTH_FILTER_ROOT}" ${PCL_INCLUDE_DIRS})
    target_compile_definitions(Raivas3DMatch PRIVATE SM_ENABLE_DEPTH_FILTER)
endif()
if(SM_BUILD_C_API)
    add_library(Raivas3DMatchC SHARED src/CApi.cpp)
    target_link_libraries(Raivas3DMatchC PRIVATE Raivas3DMatch)
    target_compile_definitions(Raivas3DMatchC PRIVATE SM_C_API_EXPORTS NOMINMAX)
    install(TARGETS Raivas3DMatchC RUNTIME DESTINATION bin ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
endif()
if(SM_BUILD_GUI)
    find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets Concurrent)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets Concurrent)
    get_target_property(_qmake Qt${QT_VERSION_MAJOR}::qmake IMPORTED_LOCATION)
    get_filename_component(_qt_bin "${_qmake}" DIRECTORY)
    add_executable(surface_match_demo WIN32 app/main.cpp app/MainWindow.cpp app/MainWindow.h app/MainWindow.ui)
    set_target_properties(surface_match_demo PROPERTIES AUTOUIC ON AUTOGEN_PARALLEL 1)
    target_link_libraries(surface_match_demo PRIVATE Raivas3DMatch Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Concurrent)
    if(SM_ENABLE_PCL_VIEWER)
        if(NOT WIN32)
            message(FATAL_ERROR "当前 PCL 嵌入窗口使用 Windows 原生 OpenGL；其他平台请设置 SM_ENABLE_PCL_VIEWER=OFF。")
        endif()
        add_library(surface_match_cloud_view STATIC app/CloudView.cpp app/CloudView.h)
        target_include_directories(surface_match_cloud_view PUBLIC app ${PCL_INCLUDE_DIRS})
        target_link_libraries(surface_match_cloud_view PUBLIC Raivas3DMatch Qt${QT_VERSION_MAJOR}::Widgets ${PCL_LIBRARIES})
        target_compile_definitions(surface_match_cloud_view PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
        target_link_libraries(surface_match_demo PRIVATE surface_match_cloud_view)
        target_compile_definitions(surface_match_demo PRIVATE SM_ENABLE_PCL_VIEWER)
        vtk_module_autoinit(TARGETS surface_match_cloud_view MODULES VTK::RenderingOpenGL2 VTK::InteractionStyle VTK::RenderingFreeType)
        get_filename_component(_compiler_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
        find_program(SM_DUMPBIN dumpbin HINTS "${_compiler_bin}")
        # 从 PCL 配置反推它配套的 VTK 路径，避免混用系统中其他 VTK 安装。
        get_filename_component(_pcl_root "${PCL_DIR}/.." ABSOLUTE)
        get_filename_component(_vtk_root "${VTK_DIR}/../../.." ABSOLUTE)
    endif()
    target_compile_definitions(surface_match_demo PRIVATE
        SM_RAIVAS_RUNTIME_DIR="${RAIVAS_RUNTIME_DIR}" SM_DEPTH_RUNTIME_DIR="${DEPTH_FILTER_RUNTIME_DIR}")
    find_program(WINDEPLOYQT windeployqt HINTS "${_qt_bin}")
    if(WINDEPLOYQT)
        add_custom_command(TARGET surface_match_demo POST_BUILD COMMAND "${WINDEPLOYQT}" --no-translations "$<TARGET_FILE:surface_match_demo>" VERBATIM)
    endif()
endif()
if(SM_BUILD_TESTS)

    add_executable(surface_match_tests tests/AlgorithmsTest.cpp)
    target_link_libraries(surface_match_tests PRIVATE Raivas3DMatch)

    if(TARGET Raivas3DMatchC)
        add_executable(surface_match_capi_tests tests/CApiTest.cpp)
        target_include_directories(surface_match_capi_tests PRIVATE include)
        target_link_libraries(surface_match_capi_tests PRIVATE Raivas3DMatchC)
        if(MSVC)
            target_compile_options(surface_match_capi_tests PRIVATE /utf-8)
        endif()
        add_test(NAME capi_lifecycle COMMAND surface_match_capi_tests)
        set_tests_properties(capi_lifecycle PROPERTIES WORKING_DIRECTORY "$<TARGET_FILE_DIR:surface_match_capi_tests>" TIMEOUT 60)
    endif()
    if(TARGET Raivas3DMatchC AND WIN32 AND CMAKE_VS_MSBUILD_COMMAND)
        add_custom_target(surface_match_csharp_smoke ALL
            COMMAND "${CMAKE_VS_MSBUILD_COMMAND}" "${CMAKE_CURRENT_SOURCE_DIR}/bindings/csharp/InteropSmoke.csproj"
                /p:Configuration=Release "/p:OutDir=$<TARGET_FILE_DIR:Raivas3DMatchC>/" /v:minimal
            DEPENDS Raivas3DMatchC VERBATIM)
        add_test(NAME csharp_interop COMMAND "$<TARGET_FILE_DIR:Raivas3DMatchC>/SurfaceMatch.InteropSmoke.exe")
        set_tests_properties(csharp_interop PROPERTIES WORKING_DIRECTORY "$<TARGET_FILE_DIR:Raivas3DMatchC>" TIMEOUT 60)
    endif()
    if(TARGET surface_match_cloud_view)
        add_executable(surface_match_cloud_tests tests/CloudViewTest.cpp)
        target_link_libraries(surface_match_cloud_tests PRIVATE surface_match_cloud_view)
        add_dependencies(surface_match_cloud_tests surface_match_demo)
        add_test(NAME pcl_viewer COMMAND surface_match_cloud_tests)
        set_tests_properties(pcl_viewer PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=windows"
            WORKING_DIRECTORY "$<TARGET_FILE_DIR:surface_match_cloud_tests>" TIMEOUT 60)

    if(TARGET surface_match_demo)
        add_executable(surface_match_settings_tests tests/SettingsTest.cpp app/MainWindow.cpp app/MainWindow.h app/MainWindow.ui)
        set_target_properties(surface_match_settings_tests PROPERTIES AUTOUIC ON AUTOGEN_PARALLEL 1)
        target_link_libraries(surface_match_settings_tests PRIVATE Raivas3DMatch Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Concurrent)
        target_compile_definitions(surface_match_settings_tests PRIVATE SM_RAIVAS_RUNTIME_DIR="${RAIVAS_RUNTIME_DIR}" SM_DEPTH_RUNTIME_DIR="${DEPTH_FILTER_RUNTIME_DIR}")
        target_include_directories(surface_match_settings_tests PRIVATE app)
        if(TARGET surface_match_cloud_view)
            target_link_libraries(surface_match_settings_tests PRIVATE surface_match_cloud_view)
            target_compile_definitions(surface_match_settings_tests PRIVATE SM_ENABLE_PCL_VIEWER)

        add_dependencies(surface_match_settings_tests surface_match_demo)

        set_tests_properties(settings_persistence PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORM_PLUGIN_PATH=${_qt_bin}/../plugins/platforms"
            WORKING_DIRECTORY "$<TARGET_FILE_DIR:surface_match_settings_tests>" TIMEOUT 30)
        add_test(NAME gui_smoke COMMAND surface_match_demo --smoke)
        set_tests_properties(gui_smoke PROPERTIES
            ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORM_PLUGIN_PATH=${_qt_bin}/../plugins/platforms"
            WORKING_DIRECTORY "$<TARGET_FILE_DIR:surface_match_demo>" TIMEOUT 90)

            set_tests_properties(gui_native_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=windows"
                WORKING_DIRECTORY "$<TARGET_FILE_DIR:surface_match_demo>" TIMEOUT 90 RUN_SERIAL TRUE)
        endif()
    endif()
endif()
# 所有本工程程序使用相同的 HALCON/OpenCV 运行时，避免依赖机器上的 PATH 顺序。
foreach(_target surface_match_demo surface_match_tests Raivas3DMatchC)
    if(TARGET ${_target} AND WIN32)
        add_custom_command(TARGET ${_target} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${HALCON_ROOT}/bin/x64-win64/halconcpp.dll" "${HALCON_ROOT}/bin/x64-win64/halcon.dll" "$<TARGET_FILE_DIR:${_target}>" VERBATIM)
        if(TARGET opencv_world)
            add_custom_command(TARGET ${_target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:opencv_world>" "$<TARGET_FILE_DIR:${_target}>" VERBATIM)
        endif()
    endif()
endforeach()
# 先复制 HALCON/OpenCV，再递归解析点云依赖，避免首次构建时目标目录尚无算法 DLL 而部署失败。
if(TARGET surface_match_cloud_view)
    add_custom_command(TARGET surface_match_demo POST_BUILD
        COMMAND ${CMAKE_COMMAND} "-DAPP=$<TARGET_FILE:surface_match_demo>" "-DDUMPBIN=${SM_DUMPBIN}"
            "-DPCL_BIN=${_pcl_root}/bin" "-DVTK_BIN=${_vtk_root}/bin" "-DQT_BIN=${_qt_bin}"
            "-DBOOST_BIN=${_pcl_root}/3rdParty/Boost/lib"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/DeployCloudRuntime.cmake" VERBATIM)
endif()
